    # 1. 連立方程式の「左辺の係数行列 (M)」を作る
    # ========================================================
    # HQの式を展開して整理すると、HQ_00やHQ_10にかかる係数は以下の行列になります。
    M = np.array([
        [1 - c * T00 * X,  -c * T01 * Y],
        [-c * T10 * X,     1 - c * T11 * Y]
    ])

    # ========================================================
    # 2. 連立方程式の「右辺の定数ベクトル (b)」を作る
    # ========================================================
    # 定数項（WやZを含む項）を右辺にまとめます。
    b1 = np.array([2 * c * T00 * W, 2 * c * T10 * W]) # HQの1列目用
    b2 = np.array([2 * c * T01 * Z, 2 * c * T11 * Z]) # HQの2列目用

    # ========================================================
    # 3. ソルバーで一発で解く (whileループの代わり)
    # ========================================================
    # np.linalg.solve が M * h = b を満たす h を数学的に逆算します。
    h1 = np.linalg.solve(M, b1)  # h1[0] が HQ[0,0], h1[1] が HQ[1,0] になる
    h2 = np.linalg.solve(M, b2)  # h2[0] が HQ[0,1], h2[1] が HQ[1,1] になる

    # 得られた解を2x2の行列 HQ に組み立てる
    HQ = np.array([
        [h1[0], h2[0]],
        [h1[1], h2[1]]
    ])

    # ========================================================
    # 4. 求まった HQ から Q を計算する (1回だけでOK)
    # ========================================================
    Q = np.empty((2, 2))
    Q[0,0] = 2 * W + HQ[0,0] * X
    Q[0,1] =         HQ[0,1] * X
    Q[1,0] =         HQ[1,0] * Y
    Q[1,1] = 2 * Z + HQ[1,1] * Y
    
    return HQ, Q  # 必要に応じてQも返すと層相関などの計算に使えます
